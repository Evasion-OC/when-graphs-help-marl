% --- Document class ---------------------------------------------------
% sn-basic = Basic Springer Nature Reference Style (author-year citations,
% sn-basic.bst), the pairing JAAMAS (a Springer, non-Nature-Portfolio
% journal) expects; confirmed against the template user manual sec. 7.9/8.3.
% The class internally issues \bibliographystyle{sn-basic} and loads
% natbib[authoryear] for this option -- do not load natbib or set a
% bibliographystyle ourselves.
\documentclass[sn-basic,pdflatex]{sn-jnl}

% --- Packages -----------------------------------------------------------
% sn-jnl.cls itself \RequirePackage's only: fontenc[T1], geometry (fixed
% Springer page geometry -- do not reload/override), hyperref, hypcap,
% breakurl, wrapfig; and (inside the sn-basic option branch) natbib in
% author-year mode plus \bibliographystyle{sn-basic}. Re-loading any of
% those five/six risks an option clash ("already defined").
%
% Confirmed EMPIRICALLY (not just from the manual's prose, which is
% ambiguous about who loads what): graphicx, multirow, amsmath, amssymb,
% amsfonts, amsthm, xcolor, textcomp, manyfoot, booktabs, algorithm,
% algorithmicx, algpseudocode, the appendix package, and
% enumitem/subcaption/microtype are NOT auto-loaded by the class -- they
% are listed in the manual only as "packages the class depends on" and
% must be \usepackage'd by the document, exactly as the class's own
% sn-article.tex sample does. Omitting amsmath in particular breaks the
% class's own \begin{document} hook (it calls \allowdisplaybreaks), so
% this list is load-bearing, not decorative.
%
% mathrsfs and listings are in the class's dependency table too, but
% main.tex uses neither \mathscr{} nor lstlisting -- loading mathrsfs
% anyway produced spurious "Font shape U/rsfs/m/n ... not available" font
% warnings at the reduced table sizes below with zero benefit, so both are
% dropped (load what main.tex uses, not the sample's maximal set).
\usepackage{graphicx}
\usepackage{multirow}
\usepackage{amsmath,amssymb,amsfonts}
\usepackage{amsthm}
\usepackage[title]{appendix}
\usepackage{xcolor}
\usepackage{textcomp}
\usepackage{manyfoot}
\usepackage{booktabs}
\usepackage{algorithm}
\usepackage{algorithmicx}
\usepackage{algpseudocode}
% What main.tex additionally needs, beyond the class's own dependency set:
\usepackage{enumitem}   % \begin{enumerate}[leftmargin=*,...] / description[style=nextline]
                         % used throughout Sections 1, Hypotheses, etc.
% caption (pulled in by subcaption) does not recognise sn-jnl and warns
% "Unknown document class, standard defaults will be used" at load time.
% Tried compatibility=false to keep caption from patching \@makecaption at
% all: that instead raises a hard "Package caption Error: Can be used only
% in preamble" the first time \begin{subfigure} executes in the body (a
% known interaction of that option with lazy first-use subfloat setup).
% So: plain load, and the harmless-in-practice fallback the warning
% announces (verified visually against the class's own caption style --
% see build report) is accepted rather than fought further.
\usepackage{subcaption}
\usepackage[protrusion=true,expansion=false]{microtype}  % same settings main.tex used under TMLR
% bookmark: hyperref's own PDF-outline builder warns "Difference (N)
% between bookmark levels is greater than one" whenever an unnumbered
% \section*{} (Broader Impact Statement, Declarations) is followed by a
% numbered \section (an \appendix one) more than one level apart in the
% outline tree it infers. The bookmark package rebuilds outlines from
% \label-level structure instead of guessing from heading numbers, which
% resolves the level jump without renumbering or renaming any section.
\usepackage{bookmark}
% \emergencystretch (standard, non-\sloppy last-resort stretch budget for
% otherwise-unbreakable lines) and a slightly relaxed \tolerance: sn-jnl's
% single-column text (31pc =~ 372pt) is narrower than TMLR's (6.5in =
% 468pt), which pushes a handful of lines with long macro-heavy technical
% terms (\textsc{} algorithm names, citation author/venue strings the
% sn-basic bibliography style renders) into moderate-badness underfull
% territory. This does not touch text content or line-breaking decisions
% for lines that already break well; it only gives TeX a bit more room on
% the few that would otherwise break badly.
\tolerance=9999
\emergencystretch=3em
% sn-jnl.cls globally \gdef's \orcidlogo (via \includegraphics{Orcidlogo.eps},
% an asset the template package does not ship) purely so its own \orcid{URL}
% macro has something to call. The orcidlink package (self-contained, no
% external image) also names its internal macro \orcidlogo, so loading it
% after the class errors with "Command \orcidlogo already defined." Clear
% the class's definition first so orcidlink's \newcommand succeeds; we then
% use orcidlink's public \orcidlink{ID} macro, not the class's \orcid{URL}.
\let\orcidlogo\relax
\usepackage{orcidlink}   % \orcidlink{...} -- vector ORCID icon, no external
                         % image file needed (sn-jnl's own \orcid{} macro
                         % requires a missing Orcidlogo.eps asset; avoided).

% \Description{...} is an acmart-specific figure-accessibility alt-text
% macro used by the shared results/phase2.tex and results/phase4.tex
% \input files. It carries no visible content under sn-jnl either, so we
% no-op it exactly as main.tex does under tmlr.sty.
\providecommand{\Description}[1]{}
